cmake_minimum_required(VERSION 3.22)
get_filename_component(LMDJ_SOURCE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(IDF_TARGET esp32s3 CACHE STRING "Cardputer ADV target")
if(NOT IDF_TARGET STREQUAL "esp32s3")
  message(FATAL_ERROR "why: incompatible hardware target; remedy: use esp32s3")
endif()
execute_process(COMMAND git -C "${LMDJ_SOURCE_ROOT}" rev-parse HEAD
  OUTPUT_VARIABLE LMDJ_CARDPUTER_REVISION OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
set(LMDJ_JSON_ARCHIVE "${LMDJ_SOURCE_ROOT}/third_party/nlohmann-json/json-v3.12.0.tar.xz")
file(SHA256 "${LMDJ_JSON_ARCHIVE}" json_hash)
if(NOT json_hash STREQUAL "42f6e95cad6ec532fd372391373363b62a14af6d771056dbfc86160e6dfff7aa")
  message(FATAL_ERROR "why: vendored JSON archive identity mismatch; remedy: restore the pinned archive")
endif()
set(LMDJ_JSON_INCLUDE "${CMAKE_BINARY_DIR}/cardputer-deps/json/include")
file(ARCHIVE_EXTRACT INPUT "${LMDJ_JSON_ARCHIVE}" DESTINATION "${CMAKE_BINARY_DIR}/cardputer-deps")
set(COMPONENTS main)
set(PROJECT_VER "1.0.59.0")
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
project(lmdj-cardputer-host)
if(CONFIG_SPIRAM)
  message(FATAL_ERROR "why: Cardputer ADV has no PSRAM; remedy: disable PSRAM and validate the internal-memory profile")
endif()

# SDK warnings policy does not override strictness for repository-owned code.
idf_build_get_property(sdk_options_expression COMPILE_OPTIONS GENERATOR_EXPRESSION)
idf_build_get_property(sdk_options COMPILE_OPTIONS)
list(FILTER sdk_options EXCLUDE REGEX "^-W")
foreach(target cardputer_foundation cardputer_domain cardputer_cooker cardputer_audio cardputer_facade __idf_main)
  get_target_property(options ${target} COMPILE_OPTIONS)
  set(strict_options)
  foreach(option IN LISTS options)
    if(option STREQUAL sdk_options_expression)
      list(APPEND strict_options ${sdk_options})
    elseif(NOT option MATCHES "^-Wno-")
      list(APPEND strict_options "${option}")
    endif()
  endforeach()
  set_property(TARGET ${target} PROPERTY COMPILE_OPTIONS "${strict_options}")
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  target_include_directories(${target} SYSTEM PRIVATE "$ENV{IDF_PATH}/components/esp_libc/platform_include")
endforeach()
