set(

  "${CMAKE_CURRENT_SOURCE_DIR}/include/lmdj/providers/local_proof_success/factory.hpp"

)

    ${lmdj_local_proof_success_package_files}
)
file(
  SHA256
  "${CMAKE_CURRENT_SOURCE_DIR}/include/lmdj/providers/local_proof_success/factory.hpp"
  lmdj_local_proof_success_factory_sha256
)

  SHA256
  "${CMAKE_CURRENT_SOURCE_DIR}/module.json"
  lmdj_local_proof_success_module_sha256
)

  SHA256
  "${CMAKE_CURRENT_SOURCE_DIR}/src/provider.cpp"
  lmdj_local_proof_success_source_sha256
)

  lmdj_local_proof_success_manifest
  "{\"files\":[{\"path\":\"providers/local-proof-success/include/lmdj/providers/local_proof_success/factory.hpp\",\"sha256\":\"${lmdj_local_proof_success_factory_sha256}\"},{\"path\":\"providers/local-proof-success/module.json\",\"sha256\":\"${lmdj_local_proof_success_module_sha256}\"},{\"path\":\"providers/local-proof-success/src/provider.cpp\",\"sha256\":\"${lmdj_local_proof_success_source_sha256}\"}],\"format\":\"provider-source-package\",\"provider_id\":\"local.proof.success\",\"provider_version\":\"1.0.3\"}\n"

  lmdj_provider_local_proof_success
  PUBLIC
    lmdj::provider_sdk

  lmdj_provider_local_proof_success
  PRIVATE
    LMDJ_LOCAL_PROOF_SUCCESS_SOURCE_PACKAGE_SHA256="${lmdj_local_proof_success_package_sha256}"
