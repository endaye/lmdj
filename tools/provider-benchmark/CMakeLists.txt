cmake_minimum_required(VERSION 3.24)
project(lmdj_slice_evaluation LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(BUILD_TESTING OFF)
if(NOT IS_DIRECTORY "${LMDJ_CANDIDATE_SOURCE}/packages/provider-sdk")
  message(FATAL_ERROR "Set LMDJ_CANDIDATE_SOURCE to the fixed candidate checkout")
endif()
include("${LMDJ_CANDIDATE_SOURCE}/cmake/LmdjDependencies.cmake")
include("${LMDJ_CANDIDATE_SOURCE}/cmake/LmdjWarnings.cmake")
add_subdirectory("${LMDJ_CANDIDATE_SOURCE}/packages/foundation" foundation)
add_subdirectory("${LMDJ_CANDIDATE_SOURCE}/packages/provider-sdk" provider-sdk)
add_subdirectory("${LMDJ_CANDIDATE_SOURCE}/providers/local-sample-slice" local-sample-slice)
add_executable(slice-evaluation run_sample_slice_evaluation.cpp)
target_link_libraries(slice-evaluation PRIVATE lmdj_provider_local_sample_slice)
lmdj_target_warnings(slice-evaluation)
